--- FILE: scripts/Animal.gd ---
class_name Animal
class_name Villager
extends CharacterBody3D

var _animal_type: String = "cow"

enum AIState { WANDERING, FLEEING, DEAD, IDLE }

@export var wander_speed: float = 1.0
@export var flee_speed: float = 3.5
@export var health: int = 6
@export var max_health: int = 6
@export var detection_range: float = 8.0

var state: AIState = AIState.WANDERING
var wander_target: Vector3
var wander_timer: float = 0.0
var leg_phase: float = 0.0
var death_timer: float = 0.0
var is_dead: bool = false
var original_position: Vector3
var hit_flash_timer: float = 0.0
var is_flashing: bool = false

var body_material: StandardMaterial3D
var original_color: Color

var torso: MeshInstance3D
var head: MeshInstance3D
var left_arm: MeshInstance3D
var right_arm: MeshInstance3D
var left_leg: MeshInstance3D
var right_leg: MeshInstance3D

var player_ref: Node3D

func _ready():
	original_position = global_position
	build_procedural_model()
	pick_new_wander_target()
	var players = get_tree().get_nodes_in_group("player")
	if players.size() > 0:
		player_ref = players[0]

func build_procedural_model():
	body_material = StandardMaterial3D.new()
	body_material.albedo_color = Color(0.8, 0.7, 0.6)
	body_material.roughness = 0.9
	original_color = body_material.albedo_color
	
	# Torso
	torso = MeshInstance3D.new()
	var t_box = BoxMesh.new()
	t_box.size = Vector3(0.8, 0.9, 0.5)
	torso.mesh = t_box
	torso.material_override = body_material
	torso.position = Vector3(0, 0.9, 0)
	add_child(torso)
	
	# Head
	head = MeshInstance3D.new()
	var h_box = BoxMesh.new()
	h_box.size = Vector3(0.5, 0.5, 0.5)
	head.mesh = h_box
	head.material_override = body_material
	head.position = Vector3(0, 1.5, 0)
	add_child(head)
	
	# Arms
	var arm_size = Vector3(0.15, 0.6, 0.15)
	left_arm = create_limb(Vector3(-0.55, 0.8, 0), arm_size)
	right_arm = create_limb(Vector3(0.55, 0.8, 0), arm_size)
	
	# Legs
	var leg_size = Vector3(0.2, 0.6, 0.2)
	left_leg = create_limb(Vector3(-0.25, 0.3, 0), leg_size)
	right_leg = create_limb(Vector3(0.25, 0.3, 0), leg_size)

func create_limb(pos: Vector3, size: Vector3) -> MeshInstance3D:
	var limb = MeshInstance3D.new()
	var box = BoxMesh.new()
	box.size = size
	limb.mesh = box
	limb.material_override = body_material
	limb.position = pos
	add_child(limb)
	return limb

func _physics_process(delta: float):
	if is_dead:
		return
	if is_flashing:
		hit_flash_timer -= delta
		if hit_flash_timer <= 0:
			set_body_color(original_color)
			is_flashing = false
	
	match state:
		AIState.WANDERING:
			handle_wandering(delta)
		AIState.FLEEING:
			handle_fleeing(delta)
		AIState.DEAD:
			handle_death(delta)
	
	if velocity.length() > 0.1 and state != AIState.DEAD:
		leg_phase += delta * velocity.length() * 2.5
		animate_limbs(leg_phase)
	else:
		animate_limbs(0.0)

func handle_wandering(delta: float):
	var direction = (wander_target - global_position).normalized()
	direction.y = 0
	var distance = global_position.distance_to(wander_target)
	if distance < 0.8:
		pick_new_wander_target()
		return
	velocity = direction * wander_speed
	move_and_slide()
	if velocity.length() > 0.1:
		var target_rot = atan2(direction.x, direction.z)
		rotation.y = lerp_angle(rotation.y, target_rot, delta * 4.0)
	wander_timer -= delta
	if wander_timer <= 0:
		pick_new_wander_target()
	if player_ref:
		var dist = global_position.distance_to(player_ref.global_position)
		if dist < detection_range:
			state = AIState.FLEEING

func handle_fleeing(delta: float):
	if not player_ref:
		state = AIState.WANDERING
		return
	var flee_dir = (global_position - player_ref.global_position).normalized()
	flee_dir.y = 0
	if flee_dir.length() < 0.1:
		flee_dir = Vector3(0,0,1)
	velocity = flee_dir * flee_speed
	move_and_slide()
	if velocity.length() > 0.1:
		var target_rot = atan2(flee_dir.x, flee_dir.z)
		rotation.y = lerp_angle(rotation.y, target_rot, delta * 5.0)
	if player_ref:
		var dist = global_position.distance_to(player_ref.global_position)
		if dist > detection_range * 1.8:
			state = AIState.WANDERING
			pick_new_wander_target()

func handle_death(delta: float):
	death_timer += delta
	rotation.x = lerp_angle(rotation.x, deg_to_rad(-90.0), delta * 3.0)
	position.y = original_position.y - 0.3
	if death_timer > 2.0:
		queue_free()

func animate_limbs(phase: float):
	var swing = sin(phase) * 0.3
	if left_leg: left_leg.rotation.x = swing
	if right_leg: right_leg.rotation.x = -swing
	if left_arm: left_arm.rotation.x = -swing * 0.5
	if right_arm: right_arm.rotation.x = swing * 0.5

func pick_new_wander_target():
	var range_ = 6.0
	wander_target = global_position + Vector3(randf_range(-range_, range_), 0, randf_range(-range_, range_))
	wander_timer = randf_range(3.0, 8.0)

func set_body_color(color: Color):
	body_material.albedo_color = color
	for node in [torso, head, left_arm, right_arm, left_leg, right_leg]:
		if node:
			node.material_override = body_material

func take_damage(amount: int):
	if is_dead:
		return
	health -= amount
	is_flashing = true
	hit_flash_timer = 0.15
	set_body_color(Color(1,0.2,0.2))
	SoundManager.play_hit()
	if health <= 0:
		die()
	else:
		state = AIState.FLEEING
		if player_ref:
			var flee_dir = (global_position - player_ref.global_position).normalized()
			wander_target = global_position + flee_dir * 8.0

func die():
	is_dead = true
	state = AIState.DEAD
	death_timer = 0.0
	velocity = Vector3.ZERO
	spawn_drop_item()
	SoundManager.play_block_break()

func spawn_drop_item():
	var drop = MeshInstance3D.new()
	var box = BoxMesh.new()
	box.size = Vector3(0.3, 0.3, 0.3)
	drop.mesh = box
	var mat = StandardMaterial3D.new()
	mat.albedo_color = Color(0.9, 0.8, 0.5)
	mat.roughness = 0.6
	drop.material_override = mat
	drop.position = global_position + Vector3(0, 0.5, 0)
	get_parent().add_child(drop)

func lerp_angle(from: float, to: float, weight: float) -> float:
	var diff = to - from
	while diff > PI: diff -= TAU
	while diff < -PI: diff += TAU
	return from + diff * weight

# ============================================================
# INPUT
# ============================================================
func _input(event: InputEvent) -> void:
	if _animating:
		return
	if event is InputEventKey and event.pressed and not event.echo:
		match event.keycode:
			KEY_RIGHT, KEY_DOWN, KEY_SPACE, KEY_ENTER, KEY_PAGEDOWN:
				if current_index < slides.size() - 1:
					show_slide(current_index + 1, true)
				else:
					show_slide(0, true)
			KEY_LEFT, KEY_UP, KEY_PAGEUP, KEY_BACKSPACE:
				if current_index > 0:
					show_slide(current_index - 1, true)
				else:
					show_slide(slides.size() - 1, true)
			KEY_HOME:
				show_slide(0, true)
			KEY_END:
				show_slide(slides.size() - 1, true)
			KEY_F:
				var w := get_window()
				if w:
					w.mode = Window.MODE_FULLSCREEN if w.mode != Window.MODE_FULLSCREEN else Window.MODE_WINDOWED
			KEY_ESCAPE:
				show_slide(0, true)

func _process(delta: float) -> void:
	# Ambient pulse on green glow
	_pulse_t += delta
	if accent_glow:
		var a : float = 0.05 + 0.035 * (0.5 + 0.5 * sin(_pulse_t * 1.6))
		accent_glow.color = Color(THEME.GREEN.r, THEME.GREEN.g, THEME.GREEN.b, a)

func show_slide(index: int, animate: bool) -> void:
	if index < 0 or index >= slides.size():
		return
	current_index = index
	_animating = true
	_sync_chrome()

	# Fade out old content
	if slide_root.get_child_count() > 0:
		var t_out := create_tween()
		for ch in slide_root.get_children():
			if ch is CanvasItem:
				t_out.parallel().tween_property(ch, "modulate:a", 0.0, THEME.ANIM_SLIDE_FADE * 0.6)
		await t_out.finished

	# Clear old
	for ch in slide_root.get_children():
		ch.queue_free()
	await get_tree().process_frame

	# Build new slide
	var data: Dictionary = slides[index]
	_build_slide_content(data)

	# Entrance fade
	var t_in := create_tween()
	for ch in slide_root.get_children():
		if ch is CanvasItem:
			ch.modulate.a = 0.0
			t_in.parallel().tween_property(ch, "modulate:a", 1.0, THEME.ANIM_SLIDE_FADE)
	await t_in.finished

	# Staggered bullets + animation hooks
	_animate_entrance(data)
	await get_tree().create_timer(0.4).timeout
	_animating = false

func _sync_chrome() -> void:
	if slide_num_label:
		slide_num_label.text = "%02d / %02d" % [current_index + 1, slides.size()]
	if progress_fill:
		var ratio : float = float(current_index + 1) / float(slides.size())
		var vw : float = size.x
		if vw <= 0:
			vw = 1280.0
		progress_fill.custom_minimum_size = Vector2(vw * ratio, 4)
		progress_fill.size = Vector2(vw * ratio, 4)

# ANIMATION LAYER — staggered bullets, table cells, pulse
# ============================================================
func _animate_entrance(data: Dictionary) -> void:
	var delay : float = 0.0
	# Fade in all bullet rows with stagger
	for ch in slide_root.get_children():
		if ch is VBoxContainer:
			for row in ch.get_children():
				if row.name.begins_with("bullet_"):
					var t := create_tween()
					t.tween_interval(delay)
					t.tween_property(row, "modulate:a", 1.0, THEME.ANIM_BULLET_FADE)
					# Slight slide up
					var start_x : float = row.position.x
					row.position.x = start_x - 18
					t.parallel().tween_property(row, "position:x", start_x, THEME.ANIM_BULLET_FADE)\
						.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)
					delay += THEME.ANIM_BULLET_DELAY

	# Fade in table cells with stagger
	var stagger : float = 0.0
	for ch in slide_root.get_children():
		if ch is GridContainer:
			for cell in ch.get_children():
				if cell.name.begins_with("cell_") or (cell is Label):
					var t := create_tween()
					t.tween_interval(stagger)
					t.tween_property(cell, "modulate:a", 1.0, THEME.ANIM_BULLET_FADE)
					stagger += 0.05

	# Title drop-in animation
	for ch in slide_root.get_children():
		if ch is Label and ch.text == data.get("title", ""):
			var start_y : float = ch.position.y
			ch.position.y = start_y - 24
			var t := create_tween()
			t.tween_property(ch, "position:y", start_y, THEME.ANIM_TITLE_DROP)\
				.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)

func play_tone(start_freq: float, end_freq: float, duration: float):
    var player = AudioStreamPlayer.new()
    add_child(player)
    var stream = AudioStreamGenerator.new()
    stream.mix_rate = sample_hz
    stream.buffer_length = duration
    player.stream = stream
    player.play()
    var playback = player.get_stream_playback()
    var total_frames = int(sample_hz * duration)
    for i in range(total_frames):
        var t = float(i) / sample_hz
        var freq = lerp(start_freq, end_freq, float(i) / total_frames)
        var phase = fmod(t * freq, 1.0)
        var sample = sin(phase * TAU) * (1.0 - float(i) / total_frames) * 0.4
        playback.push_frame(Vector2(sample, sample))
    await get_tree().create_timer(duration + 0.1).timeout
    player.queue_free()

class_name WorldGenerator
extends Node3D

@export var world_size: int = 32
@export var chunk_height: int = 16
var noise = FastNoiseLite.new()
var tree_noise = FastNoiseLite.new()
var block_meshes: Dictionary = {}

func _ready():
    noise.seed = randi()
    noise.frequency = 0.04
    tree_noise.seed = randi()
    tree_noise.frequency = 0.02
    generate_terrain()

func generate_terrain():
    for x in range(-world_size / 2, world_size / 2):
        for z in range(-world_size / 2, world_size / 2):
            var h = int(abs(noise.get_noise_2d(x, z)) * 12) + 2
            for y in range(h):
                var block_type = "grass" if y == h-1 else "dirt"
                if y == 0:
                    block_type = "stone"
                elif y == h-1 and h < 4:
                    block_type = "sand"
                create_voxel(Vector3(x, y, z), block_type)
            # Place trees
            if h > 4 and tree_noise.get_noise_2d(x, z) > 0.5:
                place_tree(Vector3(x, h, z))

func create_voxel(pos: Vector3, block_type: String):
    var mesh_instance = MeshInstance3D.new()
    var box = BoxMesh.new()
    box.size = Vector3(1, 1, 1)
    mesh_instance.mesh = box
    mesh_instance.position = pos
    
    var col_body = StaticBody3D.new()
    var col_shape = CollisionShape3D.new()
    var shape = BoxShape3D.new()
    shape.size = Vector3(1, 1, 1)
    col_shape.shape = shape
    col_body.add_child(col_shape)
    mesh_instance.add_child(col_body)
    
    var mat = StandardMaterial3D.new()
    mat.roughness = 0.9
    mat.metallic = 0.0
    match block_type:
        "grass":
            mat.albedo_color = Color(0.12, 0.35, 0.08)
        "dirt":
            mat.albedo_color = Color(0.25, 0.15, 0.08)
        "stone":
            mat.albedo_color = Color(0.35, 0.35, 0.40)
        "sand":
            mat.albedo_color = Color(0.70, 0.65, 0.50)
        "water":
            mat.albedo_color = Color(0.05, 0.15, 0.40, 0.6)
            mat.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
        "wood":
            mat.albedo_color = Color(0.25, 0.12, 0.04)
        "leaves":
            mat.albedo_color = Color(0.02, 0.20, 0.02, 0.8)
            mat.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
        _:
            mat.albedo_color = Color(0.5, 0.5, 0.5)
    mesh_instance.material_override = mat
    add_child(mesh_instance)
    block_meshes[pos] = mesh_instance

func place_tree(base_pos: Vector3):
    var trunk_height = randi_range(4, 6)
    for y in range(trunk_height):
        create_voxel(Vector3(base_pos.x, base_pos.y + y, base_pos.z), "wood")
    var leaf_start = base_pos.y + trunk_height - 2
    for dx in range(-2, 3):
        for dz in range(-2, 3):
            for dy in range(leaf_start, base_pos.y + trunk_height + 1):
                if abs(dx) == 2 and abs(dz) == 2 and dy == leaf_start + 1:
                    continue
                if abs(dx) + abs(dz) > 3:
                    continue
                var pos = Vector3(base_pos.x + dx, dy, base_pos.z + dz)
                if not block_meshes.has(pos):
                    create_voxel(pos, "leaves")

func break_voxel(hit_pos: Vector3):
    var grid_pos = (hit_pos + Vector3(0.5, 0.5, 0.5)).floor()
    var key = grid_pos
    if block_meshes.has(key):
        block_meshes[key].queue_free()
        block_meshes.erase(key)

func place_voxel(hit_pos: Vector3, block_id: int):
    var grid_pos = (hit_pos + Vector3(0.5, 0.5, 0.5)).floor()
    var type_map = {
        1: "grass",
        2: "dirt",
        3: "stone",
        4: "wood",
        5: "leaves",
        6: "sand",
        7: "water"
    }
    var block_type = type_map.get(block_id, "grass")
    if not block_meshes.has(grid_pos):
        create_voxel(grid_pos, block_type)

func get_block_type(pos: Vector3) -> String:
    var key = pos.floor()
    if block_meshes.has(key):
        var mesh = block_meshes[key]
        if mesh and mesh.material_override:
            var col = mesh.material_override.albedo_color
            if col == Color(0.12, 0.35, 0.08): return "grass"
            if col == Color(0.25, 0.15, 0.08): return "dirt"
            if col == Color(0.35, 0.35, 0.40): return "stone"
            if col == Color(0.70, 0.65, 0.50): return "sand"
            if col == Color(0.05, 0.15, 0.40, 0.6): return "water"
            if col == Color(0.25, 0.12, 0.04): return "wood"
            if col == Color(0.02, 0.20, 0.02, 0.8): return "leaves"
    return ""

# Auto-build shelter (B key)
func auto_build_shelter(center: Vector3):
    var bx = center.x
    var by = center.y
    var bz = center.z
    for x in range(bx - 3, bx + 4):
        for z in range(bz - 3, bz + 4):
            var floor_pos = Vector3(x, by - 1, z)
            if not block_meshes.has(floor_pos):
                create_voxel(floor_pos, "stone")
            if x == bx - 3 or x == bx + 3 or z == bz - 3 or z == bz + 3:
                for y in range(by, by + 3):
                    var wall_pos = Vector3(x, y, z)
                    if not block_meshes.has(wall_pos):
                        create_voxel(wall_pos, "stone")
            # Roof
            var roof_pos = Vector3(x, by + 3, z)
            if not block_meshes.has(roof_pos):
                create_voxel(roof_pos, "stone")
    # Door
    var door_pos = Vector3(bx, by, bz - 3)
    if block_meshes.has(door_pos):
        block_meshes[door_pos].queue_free()
        block_meshes.erase(door_pos)
    var door_pos2 = Vector3(bx, by + 1, bz - 3)
    if block_meshes.has(door_pos2):
        block_meshes[door_pos2].queue_free()
        block_meshes.erase(door_pos2)

# Plant tree (T key)
func plant_tree(pos: Vector3):
    var floor_pos = pos.floor()
    var h = 1
    while block_meshes.has(Vector3(floor_pos.x, floor_pos.y + h, floor_pos.z)):
        h += 1
    place_tree(Vector3(floor_pos.x, floor_pos.y + h, floor_pos.z))

func _physics_process(delta):
    if not is_on_floor():
        velocity.y -= gravity * delta

    # Jump
    if Input.is_action_just_pressed("jump") and is_on_floor():
        velocity.y = jump_velocity
        if has_node("/root/SoundManager"): get_node("/root/SoundManager").play_jump()

    # Sprint & Crouch
    is_sprinting = Input.is_key_pressed(KEY_SHIFT) and not is_crouching
    if Input.is_key_pressed(KEY_CTRL):
        is_crouching = true
    else:
        is_crouching = false

    var move_speed = speed
    if is_sprinting:
        move_speed = sprint_speed
    if is_crouching:
        move_speed *= 0.5

    var input_dir = Vector2.ZERO
    if Input.is_action_pressed("move_forward") or Input.is_key_pressed(KEY_W): input_dir.y -= 1
    if Input.is_action_pressed("move_backward") or Input.is_key_pressed(KEY_S): input_dir.y += 1
    if Input.is_action_pressed("move_left") or Input.is_key_pressed(KEY_A): input_dir.x -= 1
    if Input.is_action_pressed("move_right") or Input.is_key_pressed(KEY_D): input_dir.x += 1
    input_dir = input_dir.normalized()

    var direction = (transform.basis * Vector3(input_dir.x, 0, input_dir.y)).normalized()
    if direction:
        velocity.x = direction.x * move_speed
        velocity.z = direction.z * move_speed
    else:
        velocity.x = move_toward(velocity.x, 0, move_speed)
        velocity.z = move_toward(velocity.z, 0, move_speed)

    move_and_slide()

    # Block interaction (Left Click: Break, Right Click: Place)
    if Input.is_mouse_button_pressed(MOUSE_BUTTON_LEFT) and raycast and raycast.is_colliding():
        if attack_timer <= 0:
            var target = raycast.get_collider()
            if target and target.has_method("break_voxel"):
                target.break_voxel(raycast.get_collision_point() - raycast.get_collision_normal() * 0.4)
                if has_node("/root/SoundManager"): get_node("/root/SoundManager").play_block_break()
                play_swing_animation()
                attack_timer = attack_cooldown

    if Input.is_mouse_button_pressed(MOUSE_BUTTON_RIGHT) and raycast and raycast.is_colliding():
        if attack_timer <= 0:
            var target = raycast.get_collider()
            if target and target.has_method("place_voxel"):
                target.place_voxel(raycast.get_collision_point() + raycast.get_collision_normal() * 0.4, current_selected_block)
                if has_node("/root/SoundManager"): get_node("/root/SoundManager").play_block_place()
                play_swing_animation()
                attack_timer = attack_cooldown

    if attack_timer > 0:
        attack_timer -= delta

    # Hunger drain (every 10 seconds)
    if Engine.get_process_frames() % 600 == 0:
        hunger = max(0, hunger - 1)
        if hunger <= 0:
            health = max(0, health - 1)
        emit_signal("hunger_changed", hunger)
        emit_signal("health_changed", health)

        tween.tween_property(arm_pivot, "rotation_degrees:x", 0.0, 0.14)
