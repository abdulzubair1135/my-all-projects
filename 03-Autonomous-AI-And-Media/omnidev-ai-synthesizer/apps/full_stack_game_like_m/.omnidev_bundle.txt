func get_animal_color() -> Color:
	match animal_type:
		AnimalType.COW:
			return Color(0.55, 0.3, 0.2)
		AnimalType.SHEEP:
			return Color(0.85, 0.82, 0.78)
		AnimalType.PIG:
			return Color(0.8, 0.55, 0.55)
		_:
			return Color(0.7, 0.7, 0.7)

func _physics_process(delta: float):
	if is_dead:
		return
	
	if is_flashing:
		hit_flash_timer -= delta
		if hit_flash_timer <= 0.0:
			set_body_color(original_color)
			is_flashing = false
	
	match state:
		AIState.WANDERING:
			handle_wandering(delta)
		AIState.FLEEING:
			handle_fleeing(delta)
		AIState.DEAD:
			handle_death(delta)
		AIState.IDLE:
			handle_idle(delta)
	
	if velocity.length() > 0.1 and state != AIState.DEAD:
		leg_phase += delta * velocity.length() * 2.5
		animate_legs(leg_phase)
	else:
		animate_legs(0.0)

func handle_wandering(delta: float):
	if is_grazing:
		graze_timer -= delta
		if graze_timer <= 0.0:
			is_grazing = false
			pick_new_wander_target()
		velocity = Vector3.ZERO
		return
	
	if is_idle:
		idle_timer -= delta
		if idle_timer <= 0.0:
			is_idle = false
			pick_new_wander_target()
		velocity = Vector3.ZERO
		return
	
	var direction = (wander_target - global_position).normalized()
	direction.y = 0.0
	var distance = global_position.distance_to(wander_target)
	
	if distance < 0.8:
		if randf() < 0.4:
			is_idle = true
			idle_timer = randf_range(0.5, 2.0)
		else:
			pick_new_wander_target()
		return
	
	velocity = direction * wander_speed
	move_and_slide()
	
	if velocity.length() > 0.1:
		var target_rotation = atan2(direction.x, direction.z)
		rotation.y = lerp_angle(rotation.y, target_rotation, delta * 4.0)
	
	wander_timer -= delta
	if wander_timer <= 0.0 and not is_grazing and not is_idle:
		if randf() < 0.35:
			is_grazing = true
			graze_timer = randf_range(1.5, 4.0)
		else:
			pick_new_wander_target()
	
	if player_ref:
		var distance_to_player = global_position.distance_to(player_ref.global_position)
		if distance_to_player < detection_range:
			state = AIState.FLEEING

func handle_fleeing(delta: float):
	if not player_ref:
		state = AIState.WANDERING
		return
	
	var flee_direction = (global_position - player_ref.global_position).normalized()
	flee_direction.y = 0.0
	
	if flee_direction.length() < 0.1:
		flee_direction = Vector3(0, 0, 1)
	
	velocity = flee_direction * flee_speed
	move_and_slide()
	
	if velocity.length() > 0.1:
		var target_rotation = atan2(flee_direction.x, flee_direction.z)
		rotation.y = lerp_angle(rotation.y, target_rotation, delta * 5.0)
	
	if player_ref:
		var distance = global_position.distance_to(player_ref.global_position)
		if distance > detection_range * 1.8:
			state = AIState.WANDERING
			pick_new_wander_target()

func handle_idle(delta: float):
	idle_timer -= delta
	if idle_timer <= 0.0:
		is_idle = false
		pick_new_wander_target()

func handle_death(delta: float):
	death_timer += delta
	var target_rotation_x = deg_to_rad(-90.0)
	rotation.x = lerp_angle(rotation.x, target_rotation_x, delta * 3.0)
	position.y = original_position.y - 0.3
	
	if death_timer > 1.5:
		queue_free()

func animate_legs(phase: float):
	var leg_swing = sin(phase) * 0.3
	
	if leg_fl:
		leg_fl.rotation.x = leg_swing
	if leg_fr:
		leg_fr.rotation.x = -leg_swing
	if leg_bl:
		leg_bl.rotation.x = -leg_swing
	if leg_br:
		leg_br.rotation.x = leg_swing

func pick_new_wander_target():
	var range_x = randf_range(-8.0, 8.0)
	var range_z = randf_range(-8.0, 8.0)
	wander_target = global_position + Vector3(range_x, 0, range_z)
	wander_timer = randf_range(2.0, 6.0)
	is_grazing = false
	is_idle = false

func take_damage(amount: int):
	if is_dead:
		return
	
	health -= amount
	
	is_flashing = true
	hit_flash_timer = 0.15
	set_body_color(Color(1.0, 0.2, 0.2))
	
	if has_node("/root/SoundManager"):
		var sm = get_node("/root/SoundManager")
		if sm.has_method("play_hit"):
			sm.play_hit()
	
	if health <= 0:
		die()
	else:
		state = AIState.FLEEING
		if player_ref:
			var flee_dir = (global_position - player_ref.global_position).normalized()
			wander_target = global_position + flee_dir * 8.0

func die():

	death_timer = 0.0

	if has_node("/root/SoundManager"):

func lerp_angle(from: float, to: float, weight: float) -> float:
	var diff = to - from
	while diff > PI:
		diff -= TAU
	while diff < -PI:
		diff += TAU
	return from + diff * weight



--- FILE: scripts/SoundManager.gd ---
extends Node

var audio_player: AudioStreamPlayer
signal play_sound(sound_name: String)

func _ready():
	audio_player = AudioStreamPlayer.new()
	audio_player.name = "AudioPlayer"
	audio_player.volume_db = -8.0
	add_child(audio_player)

func _create_wav_sample(sample_count: int, sample_rate: int, generator_func: Callable) -> AudioStreamWAV:
	var wav = AudioStreamWAV.new()
	wav.format = AudioStreamWAV.FORMAT_8_BITS
	wav.mix_rate = sample_rate
	wav.stereo = false
	
	var buffer = PackedByteArray()
	buffer.resize(sample_count)
	
	for i in range(sample_count):
		var t = float(i) / float(sample_rate)
		var val = clampf(generator_func.call(i, t), -1.0, 1.0)
		var byte_val = int(round((val + 1.0) * 127.5))
		buffer[i] = clampi(byte_val, 0, 255)
		
	wav.data = buffer
	return wav

func _play_stream(stream: AudioStream):
	if audio_player:
		audio_player.stream = stream
		audio_player.play()

func play_jump():
	var sample_rate = 22050
	var total_samples = int(sample_rate * 0.12)
	var stream = _create_wav_sample(total_samples, sample_rate, func(i: int, t: float) -> float:
		var freq = lerpf(200.0, 500.0, float(i) / float(total_samples))
		var envelope = exp(-t * 12.0)
		var tone = sin(t * freq * TAU)
		var noise = randf_range(-0.15, 0.15)
		return (tone * 0.7 + noise * 0.3) * envelope * 0.6
	)
	_play_stream(stream)
	emit_signal("play_sound", "jump")

func play_break():
	var sample_rate = 22050
	var total_samples = int(sample_rate * 0.15)
	var stream = _create_wav_sample(total_samples, sample_rate, func(i: int, t: float) -> float:
		var envelope = exp(-t * 16.0)
		var crackle = randf_range(-0.7, 0.7) if (i % 3 == 0) else 0.0
		var sub_noise = randf_range(-0.2, 0.2)
		return (crackle * 0.7 + sub_noise * 0.3) * envelope * 0.7
	)
	_play_stream(stream)
	emit_signal("play_sound", "break")

func play_place():
	var sample_rate = 22050
	var total_samples = int(sample_rate * 0.1)
	var stream = _create_wav_sample(total_samples, sample_rate, func(_i: int, t: float) -> float:
		var freq = lerpf(160.0, 70.0, t / 0.1)
		var envelope = exp(-t * 22.0)
		var thud = sin(t * freq * TAU)
		var click = randf_range(-0.2, 0.2) * exp(-t * 40.0)
		return (thud * 0.75 + click * 0.25) * envelope * 0.8
	)
	_play_stream(stream)
	emit_signal("play_sound", "place")

func play_footstep():
	var sample_rate = 22050

		var tap = randf_range(-0.3, 0.3)
		return tap * envelope * 0.4
	)
	_play_stream(stream)
	emit_signal("play_sound", "footstep")

func play_hit():
	var sample_rate = 22050
	var total_samples = int(sample_rate * 0.12)
	var stream = _create_wav_sample(total_samples, sample_rate, func(_i: int, t: float) -> float:
		var envelope = exp(-t * 18.0)
		var impact = sin(t * 300.0 * TAU) * 0.5 + randf_range(-0.4, 0.4) * 0.5
		return impact * envelope * 0.75
	)
	_play_stream(stream)
	emit_signal("play_sound", "hit")

func play_damage():
	var sample_rate = 22050
	var total_samples = int(sample_rate * 0.2)
	var stream = _create_wav_sample(total_samples, sample_rate, func(_i: int, t: float) -> float:
		var envelope = exp(-t * 10.0)
		var grunt = sin(t * 120.0 * TAU) * 0.7 + randf_range(-0.25, 0.25) * 0.3
		return grunt * envelope * 0.85

func update_third_person_camera(delta: float, is_front: bool):
	var target = camera_target if camera_target else self
	var offset = Vector3(0, 2.5, 4.0) if !is_front else Vector3(0, 2.5, -4.0)
	# Use player's basis to orient offset relative to player
	var player_basis = global_transform.basis
	var desired_pos = global_position + player_basis * offset
	var desired_look = target.global_position

	var cam = front_camera if is_front else third_person_camera
	if cam:
		# Interpolate position
		cam.global_position = cam.global_position.lerp(desired_pos, delta * camera_smooth_speed)
		# Interpolate rotation to look at target
		var current_rot = cam.global_transform.basis.get_rotation_quaternion()
		# Use Basis.looking_at to get a quaternion looking at the target
		var look_dir = (desired_look - cam.global_position).normalized()
		var target_basis = Basis.looking_at(look_dir, Vector3.UP)
		var target_rot = target_basis.get_rotation_quaternion()
		var new_rot = current_rot.slerp(target_rot, delta * camera_smooth_speed)
		cam.global_transform.basis = Basis(new_rot)

func handle_movement(delta: float):
	var input_dir = Input.get_vector("move_left", "move_right", "move_forward", "move_back")
	var cam = get_active_camera()
	var forward = -cam.global_transform.basis.z if cam else -transform.basis.z
	var right = cam.global_transform.basis.x if cam else transform.basis.x

	forward.y = 0.0
	right.y = 0.0
	forward = forward.normalized()
	right = right.normalized()

	var direction = (forward * -input_dir.y + right * input_dir.x).normalized()

	var speed = walk_speed
	if Input.is_action_pressed("sprint") and !is_crouching:
		speed = sprint_speed

	if is_crouching:
		speed *= 0.5

	if not is_on_floor():
		velocity.y -= gravity * delta

	if Input.is_action_just_pressed("jump") and is_on_floor() and !is_crouching:
		velocity.y = jump_velocity
		if has_node("/root/SoundManager"):
			var sm = get_node("/root/SoundManager")
			if sm.has_method("play_jump"):
				sm.play_jump()

	if direction != Vector3.ZERO:
		velocity.x = direction.x * speed
		velocity.z = direction.z * speed
	else:
		velocity.x = move_toward(velocity.x, 0.0, speed * 0.15)
		velocity.z = move_toward(velocity.z, 0.0, speed * 0.15)

	move_and_slide()

	if velocity.length() > 0.5 and is_on_floor():
		if Engine.get_physics_frames() % 20 == 0:
			if has_node("/root/SoundManager"):
				var sm = get_node("/root/SoundManager")
				if sm.has_method("play_footstep"):
					sm.play_footstep()

func handle_block_actions(_delta: float):
	if inventory_open or !mouse_captured:
		return

	var world_gen = get_node_or_null("/root/Main/WorldGenerator")
	if !world_gen and has_node("/root/WorldGenerator"):
		world_gen = get_node("/root/WorldGenerator")
	if !world_gen:
		return

	var camera = get_active_camera()
	if !camera:
		return

	var space_state = get_world_3d().direct_space_state
	var ray_start = camera.global_position
	var ray_end = ray_start + -camera.global_transform.basis.z * block_ray_distance
	var query = PhysicsRayQueryParameters3D.create(ray_start, ray_end)
	query.exclude = [self]
	query.collision_mask = 1

	var result = space_state.intersect_ray(query)

	if result:
		var hit_pos = result.position
		var hit_normal = result.normal
		var target = result.collider

		if Input.is_action_just_pressed("break_block") or Input.is_mouse_button_pressed(MOUSE_BUTTON_LEFT):
			if attack_timer <= 0.0:
				play_swing_animation()
				if target and target.has_method("take_damage") and target != self:
					target.take_damage(1)
					attack_timer = attack_cooldown
					gain_xp(5)
					if has_node("/root/SoundManager"):
						var sm = get_node("/root/SoundManager")
						if sm.has_method("play_hit"):
							sm.play_hit()
				else:
					var block_pos = (hit_pos - hit_normal * 0.1).floor()
					if world_gen.get_block(block_pos) != 0:
						world_gen.set_block(block_pos, 0)
						attack_timer = 0.2

func generate_texture_atlas():

		5: {"base": Color(0.08, 0.38, 0.08), "pattern": "leaves"},   # LEAVES
		6: {"base": Color(0.65, 0.55, 0.35), "pattern": "plank"},    # PLANK
		7: {"base": Color(0.65, 0.8, 0.9, 0.9), "pattern": "glass"}, # GLASS (slightly transparent)
		8: {"base": Color(0.45, 0.28, 0.10), "pattern": "chest"},    # CHEST
		9: {"base": Color(0.28, 0.28, 0.3), "pattern": "cobblestone"},# COBBLESTONE
		10: {"base": Color(0.8, 0.75, 0.55), "pattern": "sand"},      # SAND

	}

	# We'll create a mapping from block_type and face normal to tile index.
	# We'll use a dict: tile_rects[(block_type, normal)] = Rect2
	# For simplicity, we'll map block_type to a tile index, but for wood we'll use two tiles.
	# We'll store tile_index for each combination.
	
	var tile_index = 0
	for block_type in range(1, 14):  # 1..13
		var base_color = tile_data.get(block_type, {"base": Color(0.5,0.5,0.5)})["base"]
		var pattern = tile_data.get(block_type, {"pattern": "none"})["pattern"]

			tile_rects[(block_type, Vector3.LEFT)] = get_tile_rect(tile_index)
			tile_rects[(block_type, Vector3.FORWARD)] = get_tile_rect(tile_index)

		else:

			for norm in [Vector3.UP, Vector3.DOWN, Vector3.RIGHT, Vector3.LEFT, Vector3.FORWARD, Vector3.BACK]:

func generate_chunk(cx: int, cz: int):
	var key = str(cx) + "," + str(cz)
	if chunk_data.has(key):
		return

	var blocks = {}

	for x in range(CHUNK_SIZE):
		for z in range(CHUNK_SIZE):
			var world_x = cx * CHUNK_SIZE + x
			var world_z = cz * CHUNK_SIZE + z

			var height = get_terrain_height(world_x, world_z)

			for y in range(WORLD_HEIGHT):
				if y <= height:
					var block_type = BlockType.DIRT
					if y == 0:
						block_type = BlockType.COBBLESTONE
					elif y == height:
						if height > 28:
							block_type = BlockType.SNOW
						elif height > 24:
							block_type = BlockType.STONE
						elif height > 16:
							block_type = BlockType.GRASS
						elif height > 10:
							block_type = BlockType.GRASS
						else:
							block_type = BlockType.SAND
					elif y >= height - 3:
						block_type = BlockType.DIRT
					else:
						block_type = BlockType.STONE

					var pos = Vector3(world_x, y, world_z)
					blocks[pos] = block_type

	generate_trees(cx, cz, blocks)

	# Surface chests
	if randf() < 0.05:
		var cx_pos = cx * CHUNK_SIZE + randi_range(3, CHUNK_SIZE - 4)
		var cz_pos = cz * CHUNK_SIZE + randi_range(3, CHUNK_SIZE - 4)
		var cy_pos = get_terrain_height(cx_pos, cz_pos) + 1
		if cy_pos > 3 and cy_pos < WORLD_HEIGHT - 2:
			blocks[Vector3(cx_pos, cy_pos, cz_pos)] = BlockType.CHEST

	chunk_data[key] = blocks
	rebuild_chunk(key)

func get_terrain_height(x: int, z: int) -> int:
	var base_h = noise.get_noise_2d(x, z) * 12.0 + 15.0
	var detail_h = detail_noise.get_noise_2d(x, z) * 3.5
	var height_var = height_noise.get_noise_2d(x * 0.5, z * 0.5) * 2.0
	var final_h = clamp(round(base_h + detail_h + height_var), 2.0, WORLD_HEIGHT - 10)
	return int(final_h)

func generate_trees(cx: int, cz: int, blocks: Dictionary):
	var tree_count = randi_range(5, 12)
	var placed = 0
	var attempts = 0

	while placed < tree_count and attempts < 250:
		attempts += 1
		var local_x = randi_range(2, CHUNK_SIZE - 3)
		var local_z = randi_range(2, CHUNK_SIZE - 3)
		var world_x = cx * CHUNK_SIZE + local_x
		var world_z = cz * CHUNK_SIZE + local_z
		var height = get_terrain_height(world_x, world_z)

		var surface_block = blocks.get(Vector3(world_x, height, world_z), BlockType.EMPTY)
		if (surface_block == BlockType.GRASS or surface_block == BlockType.DIRT) and height > 4 and height < WORLD_HEIGHT - 12:
			var pos = Vector3(world_x, 0, world_z)
			var too_close = false
			for tree_pos in tree_positions:
				if tree_pos.distance_to(pos) < 4.0:
					too_close = true
					break
			if !too_close:
				plant_tree_at(Vector3(world_x, height + 1, world_z), blocks)
				tree_positions.append(pos)
				placed += 1

func plant_tree_at(base_pos: Vector3, blocks: Dictionary):
	var bx = int(base_pos.x)
	var by = int(base_pos.y)
	var bz = int(base_pos.z)
	
	var tree_height = randi_range(5, 9)

	# Trunk
	for y in range(by, by + tree_height):
		blocks[Vector3(bx, y, bz)] = BlockType.WOOD

	# Leaves
	var leaf_start = by + tree_height - 4
	for dx in range(-3, 4):
		for dz in range(-3, 4):
			for dy in range(leaf_start, by + tree_height + 2):
				var dist = abs(dx) + abs(dz)
				var height_offset = dy - (by + tree_height)
				
				if dist == 0 and height_offset >= -1:
					continue
				if dist > 2 and height_offset > -2:
					continue
				if dist > 1 and height_offset > 0:
					continue
				if abs(dx) == 3 and abs(dz) == 3:
					continue
					
				var pos = Vector3(bx + dx, dy, bz + dz)
				if !blocks.has(pos):
					blocks[pos] = BlockType.LEAVES

func get_uv_for_face(block_type: int, normal: Vector3, vertex_index: int) -> Vector2:
	# Returns UV coordinate for a given vertex of a face (0..3) corresponding to corners of tile rect.
	var rect = tile_rects.get((block_type, normal))
	if rect == null:
		rect = Rect2(0,0,1,1)
	# Map vertex index to corner: 0->(0,0), 1->(1,0), 2->(1,1), 3->(0,1)
	var corners = [Vector2(0,0), Vector2(1,0), Vector2(1,1), Vector2(0,1)]
	var corner = corners[vertex_index % 4]
	return rect.position + corner * rect.size

func build_mesh_from_blocks(blocks: Dictionary) -> ArrayMesh:
	var vertices = []
	var normals = []
	var colors = []
	var uvs = []
	var indices = []

	for pos in blocks:
		var block_type = blocks[pos]
		if block_type == BlockType.EMPTY:
			continue

		var faces = [
			{"dir": Vector3.RIGHT, "normal": Vector3.RIGHT, "verts": get_face_vertices(pos, "right")},

			{"dir": Vector3.BACK, "normal": Vector3.BACK, "verts": get_face_vertices(pos, "back")}
		]

		for face in faces:
			var neighbor_pos = pos + face.dir
			var neighbor = blocks.get(neighbor_pos, BlockType.EMPTY)

			if neighbor == BlockType.EMPTY or (neighbor == BlockType.GLASS and block_type != BlockType.GLASS):
				var start_idx = vertices.size()
				var face_verts = face.verts

					shade_factor = 1.08
				elif normal == Vector3.DOWN:
					shade_factor = 0.55
				elif abs(normal.x) > 0.5:
					shade_factor = 0.82
				else:
					shade_factor = 0.88

				var face_col = Color(
					clamp(shade_factor, 0.0, 1.0),
					clamp(shade_factor, 0.0, 1.0),
					clamp(shade_factor, 0.0, 1.0),
					1.0
				)

				for v_idx in range(face_verts.size()):
					var v = face_verts[v_idx]
					vertices.append(Vector3(v.x, v.y, v.z))
					normals.append(normal)
					colors.append(face_col)
					var uv = get_uv_for_face(block_type, normal, v_idx)
					uvs.append(uv)

				indices.append(start_idx)
				indices.append(start_idx + 1)
				indices.append(start_idx + 2)
				indices.append(start_idx)
				indices.append(start_idx + 2)
				indices.append(start_idx + 3)

	var array_mesh = ArrayMesh.new()
	var arrays = []
	arrays.resize(Mesh.ARRAY_MAX)
	arrays[Mesh.ARRAY_VERTEX] = PackedVector3Array(vertices)
	arrays[Mesh.ARRAY_NORMAL] = PackedVector3Array(normals)
	arrays[Mesh.ARRAY_COLOR] = PackedColorArray(colors)
	arrays[Mesh.ARRAY_TEX_UV] = PackedVector2Array(uvs)
	arrays[Mesh.ARRAY_INDEX] = PackedInt32Array(indices)

	array_mesh.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES, arrays)
	return array_mesh

func get_face_vertices(pos: Vector3, face: String) -> Array:
	var x = pos.x
	var y = pos.y
	var z = pos.z

	match face:
		"right":
			return [Vector3(x+1, y, z), Vector3(x+1, y+1, z), Vector3(x+1, y+1, z+1), Vector3(x+1, y, z+1)]
		"left":
			return [Vector3(x, y, z+1), Vector3(x, y+1, z+1), Vector3(x, y+1, z), Vector3(x, y, z)]
		"up":
			return [Vector3(x, y+1, z+1), Vector3(x+1, y+1, z+1), Vector3(x+1, y+1, z), Vector3(x, y+1, z)]
		"down":
			return [Vector3(x, y, z), Vector3(x+1, y, z), Vector3(x+1, y, z+1), Vector3(x, y, z+1)]
		"forward":
			return [Vector3(x, y, z+1), Vector3(x+1, y+1, z+1), Vector3(x+1, y+1, z), Vector3(x, y, z)]
		"back":
			return [Vector3(x+1, y, z), Vector3(x+1, y+1, z), Vector3(x, y+1, z), Vector3(x, y, z)]
	return []

func get_block(pos: Vector3) -> int:

func rebuild_chunk(chunk_key: String):
	if mesh_instances.has(chunk_key):

	mesh_instances[chunk_key] = static_body

func auto_build_shelter(center_pos: Vector3):
	var bx = int(floor(center_pos.x))
	var by = int(floor(center_pos.y))
	var bz = int(floor(center_pos.z))

	for x in range(bx - 4, bx + 5):
		for z in range(bz - 4, bz + 5):
			set_block(Vector3(x, by - 1, z), BlockType.COBBLESTONE)
			set_block(Vector3(x, by + 4, z), BlockType.BRICK)

			if x == bx - 4 or x == bx + 4 or z == bz - 4 or z == bz + 4:
				for y in range(by, by + 4):
					if x == bx and z == bz - 4 and (y == by or y == by + 1):
						set_block(Vector3(x, y, z), BlockType.EMPTY)
					elif (x == bx or z == bz) and y == by + 2:
						set_block(Vector3(x, y, z), BlockType.GLASS)
					else:
						set_block(Vector3(x, y, z), BlockType.BRICK)

	set_block(Vector3(bx + 3, by, bz + 3), BlockType.CHEST)
	set_block(Vector3(bx - 3, by, bz - 3), BlockType.CHEST)

func spawn_initial_animals():
	if animal_scene == null:
		return

	var spawn_count = randi_range(12, 25)
	var attempts = 0

	while animals.size() < spawn_count and attempts < 500:
		attempts += 1
		var x = randf_range(-150, 150)
		var z = randf_range(-150, 150)
		var height = get_terrain_height(int(x), int(z))
		var y = float(height) + 1.0

		if y > 3 and y < WORLD_HEIGHT - 3:
			var animal = animal_scene.instantiate()
			animal.position = Vector3(x, y, z)
			var type_choice = randi() % 3
			match type_choice:
				0:
					animal.animal_type = Animal.AnimalType.COW
				1:
					animal.animal_type = Animal.AnimalType.SHEEP
				2:
					animal.animal_type = Animal.AnimalType.PIG
			add_child(animal)
			animals.append(animal)

func spawn_animals_in_chunk(cx: int, cz: int):
	if animal_scene == null:
		return

	if animals.size() >= max_animals:
		return

	var spawn_count = randi_range(1, 4)
	var attempts = 0

	for i in range(spawn_count):
		var local_x = randi_range(2, CHUNK_SIZE - 3)
		var local_z = randi_range(2, CHUNK_SIZE - 3)
		var world_x = cx * CHUNK_SIZE + local_x
		var world_z = cz * CHUNK_SIZE + local_z
		var height = get_terrain_height(world_x, world_z)
		var y = float(height) + 1.0

		if y > 3 and y < WORLD_HEIGHT - 3:
			var animal = animal_scene.instantiate()
			animal.position = Vector3(world_x, y, world_z)
			var type_choice = randi() % 3
			match type_choice:
				0:
					animal.animal_type = Animal.AnimalType.COW
				1:
					animal.animal_type = Animal.AnimalType.SHEEP
				2:
					animal.animal_type = Animal.AnimalType.PIG
			add_child(animal)
			animals.append(animal)
			break
		attempts += 1
		if attempts > 20:
			break

func update_player_position(player_pos: Vector3):
	var chunk_x = int(floor(player_pos.x / CHUNK_SIZE))
	var chunk_z = int(floor(player_pos.z / CHUNK_SIZE))

	for x in range(-6, 7):
		for z in range(-6, 7):
			var key = str(chunk_x + x) + "," + str(chunk_z + z)
			if !chunk_data.has(key):
				generate_chunk(chunk_x + x, chunk_z + z)
				spawn_animals_in_chunk(chunk_x + x, chunk_z + z)

	clear_distant_chunks(player_pos)

func clear_distant_chunks(player_pos: Vector3):
	var chunk_x = int(floor(player_pos.x / CHUNK_SIZE))
	var chunk_z = int(floor(player_pos.z / CHUNK_SIZE))

	var to_remove = []
	for key in chunk_data:
		var parts = key.split(",")
		var cx = int(parts[0])
		var cz = int(parts[1])

		if abs(cx - chunk_x) > 8 or abs(cz - chunk_z) > 8:
			to_remove.append(key)

	for key in to_remove:
		if mesh_instances.has(key):
			mesh_instances[key].queue_free()
			mesh_instances.erase(key)
		chunk_data.erase(key)

		var animals_to_remove = []
		for animal in animals:
			if is_instance_valid(animal):
				var animal_chunk_x = int(floor(animal.position.x / CHUNK_SIZE))
				var animal_chunk_z = int(floor(animal.position.z / CHUNK_SIZE))
				if str(animal_chunk_x) + "," + str(animal_chunk_z) == key:
					animal.queue_free()
					animals_to_remove.append(animal)
		for animal in animals_to_remove:
			animals.erase(animal)

			menu_btn = pause_menu.get_node_or_null("MenuButton")
		if menu_btn and !menu_btn.is_connected("pressed", Callable(self, "_on_main_menu")):
			menu_btn.pressed.connect(_on_main_menu)

		var quit_btn = pause_menu.get_node_or_null("VBoxContainer/QuitButton")
		if !quit_btn:
			quit_btn = pause_menu.get_node_or_null("QuitButton")
		if quit_btn and !quit_btn.is_connected("pressed", Callable(self, "_on_quit_pressed")):
			quit_btn.pressed.connect(_on_quit_pressed)

	if quit_menu:
		var confirm_btn = quit_menu.get_node_or_null("VBoxContainer/ConfirmButton")
		if !confirm_btn:
			confirm_btn = quit_menu.get_node_or_null("ConfirmButton")
		if confirm_btn and !confirm_btn.is_connected("pressed", Callable(self, "_on_quit_game")):
			confirm_btn.pressed.connect(_on_quit_game)

		var cancel_btn = quit_menu.get_node_or_null("VBoxContainer/CancelButton")
		if !cancel_btn:
			cancel_btn = quit_menu.get_node_or_null("CancelButton")
		if cancel_btn and !cancel_btn.is_connected("pressed", Callable(self, "_on_cancel_quit")):
			cancel_btn.pressed.connect(_on_cancel_quit)

func _on_open_character_menu():
	if start_menu:
		start_menu.visible = false
	if character_menu:
		character_menu.visible = true
		update_character_preview()

func _on_start_game():
	if start_menu:
		start_menu.visible = false
	Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)
	var current_player = get_tree().get_first_node_in_group("player")
	if current_player:
		current_player.mouse_captured = true

func _on_quit_pressed():
	quit_confirm_active = true
	quit_timer = 0.0
	if start_menu:
		start_menu.visible = false
	if pause_menu:
		pause_menu.visible = false
	if quit_menu:
		quit_menu.visible = true
		if quit_timer_label:
			quit_timer_label.text = "Quitting in " + str(quit_confirm_delay) + "s..."

func _on_cancel_quit():
	quit_confirm_active = false
	quit_timer = 0.0
	if quit_menu:
		quit_menu.visible = false
	if start_menu:
		start_menu.visible = true

func _on_quit_game():
	get_tree().quit()

func _on_resume_game():
	if pause_menu:
		pause_menu.visible = false
	Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)
	var current_player = get_tree().get_first_node_in_group("player")
	if current_player:
		current_player.mouse_captured = true
	get_tree().paused = false

func _on_main_menu():
	if pause_menu:
		pause_menu.visible = false
	if start_menu:
		start_menu.visible = true
	Input.set_mouse_mode(Input.MOUSE_MODE_VISIBLE)
	var current_player = get_tree().get_first_node_in_group("player")
	if current_player:
		current_player.mouse_captured = false
	get_tree().paused = false
	get_tree().reload_current_scene()

func _input(event: InputEvent):
	if event is InputEventKey and event.pressed and event.keycode == KEY_ESCAPE:
		if quit_menu and quit_menu.visible:
			_on_cancel_quit()
		elif character_menu and character_menu.visible:
			_on_close_character_menu()
		else:
			toggle_pause_menu()

func _process(delta: float):
	if quit_confirm_active and quit_menu and quit_menu.visible:
		quit_timer += delta
		if quit_timer_label:
			var remaining = max(0, quit_confirm_delay - quit_timer)
			quit_timer_label.text = "Quitting in " + str(int(remaining)) + "s..."
		if quit_timer >= quit_confirm_delay:
			_on_quit_game()

func toggle_pause_menu():
	if start_menu and start_menu.visible:
		return
	if character_menu and character_menu.visible:
		return

	if pause_menu:
		var current_player = get_tree().get_first_node_in_group("player")
		if pause_menu.visible:
			pause_menu.visible = false
			Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)
			if current_player:
				current_player.mouse_captured = true
			get_tree().paused = false
		else:
			pause_menu.visible = true
			Input.set_mouse_mode(Input.MOUSE_MODE_VISIBLE)
			if current_player:
				current_player.mouse_captured = false
			get_tree().paused = true

  mesh.castShadow = true;
  mesh.receiveShadow = true;
  mesh.userData = { x, y, z, typeId };
  scene.add(mesh);
  blocks.set(key, mesh);
}

function removeBlock(x, y, z) {
  const key = `${x},${y},${z}`;
  const mesh = blocks.get(key);
  if (mesh) {
    scene.remove(mesh);
    blocks.delete(key);
    Sound.break();
  }
}

// Generate 3D Voxel World
const worldSize = 24;
for (let x = -worldSize; x <= worldSize; x++) {
  for (let z = -worldSize; z <= worldSize; z++) {
    const dist = Math.hypot(x, z);
    const height = Math.floor(Math.sin(x * 0.2) * Math.cos(z * 0.2) * 3 + 3);
    for (let y = 0; y <= height; y++) {
      const type = (y === height) ? 1 : ((y > height - 2) ? 2 : 3);
      addBlock(x, y, z, type);
    }
  }
}

// Player Physics & Controls
let activeBlockId = 1;
const player = {
  pos: new THREE.Vector3(0, 10, 0),
  vel: new THREE.Vector3(),
  yaw: 0,
  pitch: 0,
  onGround: false,
  speed: 9,
  jumpVel: 11
};

const keys = {};

  keys[e.code] = true;
  if (e.key >= '1' && e.key <= '5') {
    activeBlockId = parseInt(e.key);
	document.querySelectorAll('.slot').forEach(s => s.classList.remove('active'));
	document.querySelector(`.slot[data-id="${activeBlockId}"]`).classList.add('active');
  }
});
window.addEventListener('keyup', e => keys[e.code] = false);

// Pointer Lock Controls

blocker.addEventListener('click', () => {
  document.body.requestPointerLock();
});

document.addEventListener('pointerlockchange', () => {
  isLocked = document.pointerLockElement === document.body;
  blocker.style.display = isLocked ? 'none' : 'flex';
});

document.addEventListener('mousemove', e => {
  if (!isLocked) return;
  player.yaw -= e.movementX * 0.0025;
  player.pitch -= e.movementY * 0.0025;
  player.pitch = Math.max(-Math.PI / 2.1, Math.min(Math.PI / 2.1, player.pitch));
});

// Raycasting Block Interaction
const raycaster = new THREE.Raycaster();
window.addEventListener('mousedown', e => {
  if (!isLocked) return;
  raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
  const intersects = raycaster.intersectObjects(Array.from(blocks.values()));
  
  if (intersects.length > 0 && intersects[0].distance < 6) {
    const hit = intersects[0];
    const bData = hit.object.userData;
    
    if (e.button === 0) {
      // Break Block
      removeBlock(bData.x, bData.y, bData.z);
    } else if (e.button === 2) {
      // Place Block
      const norm = hit.face.normal;
      const nx = bData.x + Math.round(norm.x);
      const ny = bData.y + Math.round(norm.y);
      const nz = bData.z + Math.round(norm.z);
      addBlock(nx, ny, nz, activeBlockId);
      Sound.place();
    }
  }
});

// Game Loop
let lastTime = performance.now();
function animate() {
  requestAnimationFrame(animate);
  const now = performance.now();
  const dt = Math.min((now - lastTime) / 1000, 0.1);
  lastTime = now;

  if (isLocked) {
    // Movement
	const moveX = (keys['KeyD'] ? 1 : 0) - (keys['KeyA'] ? 1 : 0);
	const moveZ = (keys['KeyS'] ? 1 : 0) - (keys['KeyW'] ? 1 : 0);
    
    const forward = new THREE.Vector3(-Math.sin(player.yaw), 0, -Math.cos(player.yaw));
    const right = new THREE.Vector3(Math.cos(player.yaw), 0, -Math.sin(player.yaw));
    
    const moveDir = new THREE.Vector3();
    moveDir.addScaledVector(right, moveX);
    moveDir.addScaledVector(forward, -moveZ);
    moveDir.normalize();

    player.vel.x = moveDir.x * player.speed;
    player.vel.z = moveDir.z * player.speed;

    // Gravity
    player.vel.y -= 26 * dt;
	if (keys['Space'] && player.onGround) {
      player.vel.y = player.jumpVel;
      player.onGround = false;
      Sound.jump();
    }

    // Voxel Terrain Collision (Simple Ground Detection)
    player.pos.x += player.vel.x * dt;
    player.pos.z += player.vel.z * dt;
    player.pos.y += player.vel.y * dt;

    const blockUnderX = Math.floor(player.pos.x);
    const blockUnderZ = Math.floor(player.pos.z);
    let highestY = -10;

    for (let y = 0; y < 20; y++) {
      if (blocks.has(`${blockUnderX},${y},${blockUnderZ}`)) {
        highestY = Math.max(highestY, y + 1.8);
      }
    }

    if (player.pos.y <= highestY) {
      player.pos.y = highestY;
      player.vel.y = 0;
      player.onGround = true;
    } else {
      player.onGround = false;
    }
  }

  // Update Camera
  camera.position.copy(player.pos);
  camera.rotation.set(0, 0, 0);
  camera.rotation.y = player.yaw;
  camera.rotation.x = player.pitch;

  renderer.render(scene, camera);
}

window.addEventListener('resize', () => {
  camera.aspect = window.innerWidth / window.innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(window.innerWidth, window.innerHeight);
});

animate();
